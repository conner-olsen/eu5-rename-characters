	if = {
		limit = { NOT = { exists = character:rch_storage_char_$num$ } }

			first_name = name_birger
			birth_date = 1.1.1
			culture = culture:turkish_culture
			religion = religion:catholic
			adm = 0
			dip = 0
			mil = 0

			create_in_limbo = yes
		}
		character:rch_storage_char_$num$ = {
			set_nickname = ""
		}
	}
}
