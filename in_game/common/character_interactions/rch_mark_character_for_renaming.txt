rch_mark_character_for_renaming_interaction = {
	ai_tick = never
	message = no

	select_trigger = {
		looking_for_a = character
		target_flag = recipient
		name = "choose_character"
		column = {
			data = name
		}
		visible = {
			always = yes
		}
		enabled = {
			always = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = {
				name = rch_marked_character
				value = scope:recipient
			}
			rch_set_dynasty_to_be_renamed = yes
		}
	}
}
